create table public.links (
  id bigint generated by default as identity not null,
  name character varying null,
  slug character varying not null,
  url text null,
  user_id uuid null,
  status character varying null default 'active',
  created_at timestamp with time zone not null default now(),
  constraint links_pkey primary key (id),
  constraint links_path_key unique (slug),
  constraint links_user_id_fkey foreign key (user_id) references auth.users (id)
) TABLESPACE pg_default;

create table public.bulk_batches (
  id bigint generated by default as identity not null,
  name character varying not null,
  link_count integer not null,
  status character varying not null default 'ready',
  created_at timestamp with time zone not null default now(),
  constraint bulk_batches_pkey primary key (id)
) TABLESPACE pg_default;

create table public.activity (
  id bigint generated by default as identity not null,
  link_slug character varying not null,
  location character varying null,
  device character varying null,
  type character varying not null,
  time timestamp with time zone not null default now(),
  constraint activity_pkey primary key (id)
) TABLESPACE pg_default;
